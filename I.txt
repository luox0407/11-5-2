<script>
  let questions = [];
  let answers = {};
  let arUrl = '';


  function startLearning() {
    document.getElementById('home').classList.add('hidden');
    document.getElementById('learning').classList.remove('hidden');
    loadQuestions();
  }


  function showTab(tab) {
    const isAR = tab === 'ar';


    document.getElementById('arSection').classList.toggle('hidden', !isAR);
    document.getElementById('worksheetSection').classList.toggle('hidden', isAR);


    document.getElementById('arTab').classList.toggle('active', isAR);
    document.getElementById('worksheetTab').classList.toggle('active', !isAR);
  }


  function openAR() {
    if (!arUrl) {
      alert('AR 觀察功能尚在準備中，請先完成學習單或等待教師通知。');
      return;
    }


    window.open(arUrl, '_blank');
  }


  function loadQuestions() {
    const container = document.getElementById('questions');


    google.script.run
      .withSuccessHandler(data => {
        questions = data;
        renderQuestions();
      })
      .withFailureHandler(error => {
        container.textContent = '題目載入失敗：' + error.message;
      })
      .getQuestions();
  }


  function renderQuestions() {
    const container = document.getElementById('questions');
    container.innerHTML = '';


    if (!questions.length) {
      container.textContent = '目前沒有題目，請教師檢查 Google Sheets。';
      return;
    }


    questions.forEach((q, index) => {
      const id = String(q['題目編號']);
      const type = String(q['題型'] || '');
      const question = document.createElement('div');
      question.className = 'question';


      const title = document.createElement('h3');
      title.textContent = (index + 1) + '. ' + q['題目'];
      question.appendChild(title);


      if (type.includes('選')) {
        const options = String(q['選項'] || '')
          .split(/[,，、]/)
          .map(x => x.trim())
          .filter(Boolean);


        options.forEach(option => {
          const label = document.createElement('label');
          const radio = document.createElement('input');


          radio.type = 'radio';
          radio.name = 'q_' + id;
          radio.value = option;
          radio.style.width = 'auto';


          radio.addEventListener('change', () => {
            answers[id] = option;
          });


          label.appendChild(radio);
          label.appendChild(document.createTextNode(' ' + option));
          question.appendChild(label);
        });
      } else {
        const textarea = document.createElement('textarea');
        textarea.rows = 3;
        textarea.placeholder = '請輸入你的答案';


        textarea.value = answers[id] || '';
        textarea.addEventListener('input', () => {
          answers[id] = textarea.value;
        });


        question.appendChild(textarea);
      }


      container.appendChild(question);
    });
  }


  function submitAnswers() {
    const className = document.getElementById('className').value.trim();
    const seatNumber = document.getElementById('seatNumber').value.trim();
    const studentName = document.getElementById('studentName').value.trim();
    const message = document.getElementById('message');
    const button = document.getElementById('submitButton');


    if (!className || !seatNumber || !studentName) {
      alert('請完整填寫班級、座號及姓名。');
      return;
    }


    const result = questions.map(q => ({
      questionId: String(q['題目編號']),
      answer: String(answers[String(q['題目編號'])] || '').trim()
    }));


    if (result.some(x => !x.answer)) {
      alert('還有題目尚未填答，請檢查學習單。');
      return;
    }


    button.disabled = true;
    message.textContent = '正在送出……';


    google.script.run
      .withSuccessHandler(text => {
        message.textContent = text;
        button.disabled = false;
      })
      .withFailureHandler(error => {
        message.textContent = '送出失敗：' + error.message;
        button.disabled = false;
      })
      .saveAnswers({
        className,
        seatNumber,
        studentName,
        answers: result
      });
  }
</script>
