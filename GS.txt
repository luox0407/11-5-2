const SPREADSHEET_ID = '19ty9MPoF70Jj2w6ResqFO043YDuLmg7TkZ5z9aVkKAM';


function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('水蘊草學習平台')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}


// 讀取題目
function getQuestions() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('題目');


  if (!sheet) {
    throw new Error('找不到「題目」工作表');
  }


  const values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];


  const headers = values[0];


  return values.slice(1)
    .filter(row => row[0] !== '')
    .map(row => {
      const item = {};
      headers.forEach((header, index) => {
        item[header] = row[index];
      });
      return item;
    });
}


// 儲存學生答案
function saveAnswers(data) {
  if (!data || !Array.isArray(data.answers) || !data.answers.length) {
    throw new Error('沒有可送出的答案');
  }


  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('學生作答');


  if (!sheet) {
    throw new Error('找不到「學生作答」工作表');
  }


  const rows = data.answers.map(answer => [
    new Date(),
    data.className,
    data.seatNumber,
    data.studentName,
    answer.questionId,
    answer.answer
  ]);


  const lock = LockService.getScriptLock();
  lock.waitLock(10000);


  try {
    sheet.getRange(
      sheet.getLastRow() + 1,
      1,
      rows.length,
      rows[0].length
    ).setValues(rows);
  } finally {
    lock.releaseLock();
  }


  return '作答已成功送出！';
}
